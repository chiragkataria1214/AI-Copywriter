                                  Good
                                </button>
                                <button
                                  onClick={() => rateHeadline(index, 'perfect')}
                                  className={`flex items-center gap-1 px-2 py-1 rounded-md text-xs transition-colors ${
                                    feedbackData.headlineRatings[headline] === 'perfect'
                                      ? 'bg-yellow-100 text-yellow-700'
                                      : 'bg-slate-100 text-slate-600 hover:bg-yellow-50'
                                  }`}
                                >
                                  <Star className="w-3 h-3" />
                                  Perfect
                                </button>
                                <button
                                  onClick={() => rateHeadline(index, 'bad')}
                                  className={`flex items-center gap-1 px-2 py-1 rounded-md text-xs transition-colors ${
                                    feedbackData.headlineRatings[headline] === 'bad'
                                      ? 'bg-red-100 text-red-700'
                                      : 'bg-slate-100 text-slate-600 hover:bg-red-50'
                                  }`}
                                >
                                  <ThumbsDown className="w-3 h-3" />
                                  Bad
                                </button>
                              </div>
                            </div>
                          </div>
                        ))}
                      </div>
                    </div>

                    {/* Primary Text */}
                    <div>
                      <div className="flex items-center justify-between mb-4">
                        <h3 className="text-lg font-semibold text-slate-800">Primary Text</h3>
                        <button
                          onClick={copyPrimaryText}
                          className="flex items-center gap-2 px-3 py-2 bg-slate-100 hover:bg-slate-200 rounded-lg transition-colors text-slate-700 text-sm font-medium"
                        >
                          {copiedPrimaryText ? (
                            <>
                              <Check className="w-4 h-4 text-green-600" />
                              Copied!
                            </>
                          ) : (
                            <>
                              <Copy className="w-4 h-4" />
                              Copy Primary Text
                            </>
                          )}
                        </button>
                      </div>
                      <div className="bg-white/80 rounded-xl p-4 border border-slate-200">
                        <p className="text-slate-700 leading-relaxed whitespace-pre-wrap mb-4">{generatedPrimaryText}</p>
                        <div className="flex items-center gap-2 pt-2 border-t border-slate-200">
                          <span className="text-sm text-slate-500 mr-2">Rate this copy:</span>
                          <button
                            onClick={() => ratePrimaryText('good')}
                            className={`flex items-center gap-1 px-3 py-1 rounded-md text-xs transition-colors ${
                              feedbackData.primaryTextRatings[generatedPrimaryText] === 'good'
                                ? 'bg-green-100 text-green-700'
                                : 'bg-slate-100 text-slate-600 hover:bg-green-50'
                            }`}
                          >
                            <ThumbsUp className="w-3 h-3" />
                            Good
                          </button>
                          <button
                            onClick={() => ratePrimaryText('perfect')}
                            className={`flex items-center gap-1 px-3 py-1 rounded-md text-xs transition-colors ${
                              feedbackData.primaryTextRatings[generatedPrimaryText] === 'perfect'
                                ? 'bg-yellow-100 text-yellow-700'
                                : 'bg-slate-100 text-slate-600 hover:bg-yellow-50'
                            }`}
                          >
                            <Star className="w-3 h-3" />
                            Perfect
                          </button>
                          <button
                            onClick={() => ratePrimaryText('bad')}
                            className={`flex items-center gap-1 px-3 py-1 rounded-md text-xs transition-colors ${
                              feedbackData.primaryTextRatings[generatedPrimaryText] === 'bad'
                                ? 'bg-red-100 text-red-700'
                                : 'bg-slate-100 text-slate-600 hover:bg-red-50'
                            }`}
                          >
                            <ThumbsDown className="w-3 h-3" />
                            Bad
                          </button>
                        </div>
                      </div>
                    </div>
                  </div>
                ) : (
                  <div className="flex flex-col items-center justify-center h-64 text-slate-400">
                    <Target className="w-16 h-16 mb-4 opacity-50" />
                    <p className="text-lg">Your compelling ad copy will appear here</p>
                    <p className="text-sm mt-2 text-center">Upload transcription and select persona to get started</p>
                  </div>
                )
              ) : (
                // Landing Page Copy Output
                generatedLandingCopy.headline ? (
                  <div className="space-y-6">
                    <div className="flex items-center justify-between mb-4">
                      <h3 className="text-lg font-semibold text-slate-800">
                        {landingPageTypes[landingPageType]?.label || 'Landing Page'} Copy
                      </h3>
                      <button
                        onClick={copyLandingPage}
                        className="flex items-center gap-2 px-3 py-2 bg-slate-100 hover:bg-slate-200 rounded-lg transition-colors text-slate-700 text-sm font-medium"
                      >
                        {copiedLandingCopy ? (
                          <>
                            <Check className="w-4 h-4 text-green-600" />
                            Copied!
                          </>
                        ) : (
                          <>
                            <Copy className="w-4 h-4" />
                            Copy All
                          </>
                        )}
                      </button>
                    </div>

                    <div className="bg-white/80 rounded-xl p-6 border border-slate-200 space-y-6">
                      {/* Headline */}
                      <div>
                        <h4 className="font-semibold text-slate-800 mb-2">Headline</h4>
                        <p className="text-slate-700 text-xl font-medium leading-relaxed">{generatedLandingCopy.headline}</p>
                      </div>

                      {/* Subheadline */}
                      {generatedLandingCopy.subheadline && (
                        <div>
                          <h4 className="font-semibold text-slate-800 mb-2">Subheadline</h4>
                          <p className="text-slate-600 leading-relaxed">{generatedLandingCopy.subheadline}</p>
                        </div>
                      )}

                      {/* Introduction */}
                      {generatedLandingCopy.introduction && (
                        <div>
                          <h4 className="font-semibold text-slate-800 mb-2">Introduction</h4>
                          <p className="text-slate-700 leading-relaxed whitespace-pre-wrap">{generatedLandingCopy.introduction}</p>
                        </div>
                      )}

                      {/* Sections */}
                      {generatedLandingCopy.sections.map((section, index) => (
                        <div key={index}>
                          <h4 className="font-semibold text-slate-800 mb-2">{section.title}</h4>
                          <p className="text-slate-700 leading-relaxed whitespace-pre-wrap">{section.content}</p>
                        </div>
                      ))}

                      {/* Social Proof */}
                      {generatedLandingCopy.socialProof && (
                        <div>
                          <h4 className="font-semibold text-slate-800 mb-2">Social Proof</h4>
                          <p className="text-slate-700 leading-relaxed whitespace-pre-wrap">{generatedLandingCopy.socialProof}</p>
                        </div>
                      )}

                      {/* Risk Reversal */}
                      {generatedLandingCopy.riskReversal && (
                        <div>
                          <h4 className="font-semibold text-slate-800 mb-2">Risk Reversal</h4>
                          <p className="text-slate-700 leading-relaxed whitespace-pre-wrap">{generatedLandingCopy.riskReversal}</p>
                        </div>
                      )}

                      {/* Conclusion */}
                      {generatedLandingCopy.conclusion && (
                        <div>
                          <h4 className="font-semibold text-slate-800 mb-2">Conclusion</h4>
                          <p className="text-slate-700 leading-relaxed whitespace-pre-wrap">{generatedLandingCopy.conclusion}</p>
                        </div>
                      )}

                      {/* CTA */}
                      {generatedLandingCopy.cta && (
                        <div>
                          <h4 className="font-semibold text-slate-800 mb-2">Call-to-Action</h4>
                          <p className="text-slate-700 font-medium leading-relaxed">{generatedLandingCopy.cta}</p>
                        </div>
                      )}

                      {/* Rating */}
                      <div className="flex items-center gap-2 pt-4 border-t border-slate-200">
                        <span className="text-sm text-slate-500 mr-2">Rate this landing page:</span>
                        <button
                          onClick={() => rateLandingPage('good')}
                          className={`flex items-center gap-1 px-3 py-1 rounded-md text-xs transition-colors ${
                            feedbackData.landingPageRatings[`${generatedLandingCopy.headline}_${landingPageType}`] === 'good'
                              ? 'bg-green-100 text-green-700'
                              : 'bg-slate-100 text-slate-600 hover:bg-green-50'
                          }`}
                        >
                          <ThumbsUp className="w-3 h-3" />
                          Good
                        </button>
                        <button
                          onClick={() => rateLandingPage('perfect')}
                          className={`flex items-center gap-1 px-3 py-1 rounded-md text-xs transition-colors ${
                            feedbackData.landingPageRatings[`${generatedLandingCopy.headline}_${landingPageType}`] === 'perfect'
                              ? 'bg-yellow-100 text-yellow-700'
                              : 'bg-slate-100 text-slate-600 hover:bg-yellow-50'
                          }`}
                        >
                          <Star className="w-3 h-3" />
                          Perfect
                        </button>
                        <button
                          onClick={() => rateLandingPage('bad')}
                          className={`flex items-center gap-1 px-3 py-1 rounded-md text-xs transition-colors ${
                            feedbackData.landingPageRatings[`${generatedLandingCopy.headline}_${landingPageType}`] === 'bad'
                              ? 'bg-red-100 text-red-700'
                              : 'bg-slate-100 text-slate-600 hover:bg-red-50'
                          }`}
                        >
                          <ThumbsDown className="w-3 h-3" />
                          Bad
                        </button>
                      </div>
                    </div>
                  </div>
                ) : (
                  <div className="flex flex-col items-center justify-center h-64 text-slate-400">
                    <Globe className="w-16 h-16 mb-4 opacity-50" />
                    <p className="text-lg">Your compelling landing page will appear here</p>
                    <p className="text-sm mt-2 text-center">
                      Select {landingPageTypes[landingPageType]?.label?.toLowerCase() || 'landing page'} and persona to get started
                    </p>
                  </div>
                )
              )}
            </div>

            {/* Feedback Analytics */}
            {(feedbackData.goodPatterns.length > 0 || feedbackData.badPatterns.length > 0) && (
              <div className="bg-gradient-to-r from-indigo-50 to-blue-50 rounded-2xl p-6 border border-indigo-200">
                <h3 className="font-semibold text-slate-800 mb-3">📊 Feedback Analytics</h3>
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  {feedbackData.goodPatterns.length > 0 && (
                    <div className="bg-green-50 rounded-lg p-3">
                      <h4 className="font-medium text-green-800 mb-2">✅ Learning from Good Copy</h4>
                      <div className="text-xs text-green-700 space-y-1">
                        {feedbackData.goodPatterns.slice(-3).map((pattern, index) => (
                          <div key={index}>• {pattern}</div>
                        ))}
                      </div>
                    </div>
                  )}
                  {feedbackData.badPatterns.length > 0 && (
                    <div className="bg-red-50 rounded-lg p-3">
                      <h4 className="font-medium text-red-800 mb-2">❌ Avoiding Bad Patterns</h4>
                      <div className="text-xs text-red-700 space-y-1">
                        {feedbackData.badPatterns.slice(-3).map((pattern, index) => (
                          <div key={index}>• {pattern}</div>
                        ))}
                      </div>
                    </div>
                  )}
                </div>
                <div className="mt-3 text-xs text-indigo-600">
                  Next generation will use this feedback to create better copy!
                </div>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
